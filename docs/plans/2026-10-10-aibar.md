# Plan: aibar

**Spec:** `docs/specs/aibar/SPEC.md` · **Depth:** Deep · **Tasks:** 15 · **Date:** 2026-10-10

Noctalia Shell v4 plugin showing Claude Code and Grok plan usage in the bar and a
panel. Same architecture as Daybar: logic in `utils/**/*.js` (QML JavaScript
libraries, `.pragma library`, top-level declarations at column 0, `.import` for
deps, no ES `import`/`export`), tested with vitest through the transform in
`vitest.config.js`; QML is thin.

**Global rules for every task**

- TDD for `utils/`: write the test, run it and see it fail, then implement.
  `npm test -- tests/<file>.test.js`, then `npm test` (read `Tests  N passed`).
- Never write a real token, email or `/home/<user>` path anywhere (fixtures use
  `/data/u`). Tokens only in XHR headers; never `console.log`/`Logger` them.
- English code/comments; UI strings via `pluginApi?.tr()` in both i18n files.
- No git commands inside tasks (the orchestrator commits).

**Implement** `utils/creds.js`
```js
.pragma library

// Reads the CLIs' own credential files (text in, plain result out). Never logs,
// copies or writes a token; the caller passes it straight into a request header.

var EXPIRY_MARGIN_MS = 60000;

function parse(text) {
  try {
    return JSON.parse(String(text || ""));
  } catch (e) {
    return null;
  }
}

// ~/.claude/.credentials.json -> { ok, token, plan } | { ok: false, error }
function claudeToken(text, nowMs) {
  const o = parse(text);
  const oauth = o && o.claudeAiOauth;
  if (!oauth || typeof oauth.accessToken !== "string" || oauth.accessToken === "") return { ok: false, error: "claude-login" };
  if (!(Number(oauth.expiresAt) > nowMs + EXPIRY_MARGIN_MS)) return { ok: false, error: "claude-expired" };
  return { ok: true, token: oauth.accessToken, plan: String(oauth.subscriptionType || "") };
}

// ~/.grok/auth.json -> { ok, key } | { ok: false, error }
function grokKey(text) {
  const o = parse(text);
  if (!o || typeof o !== "object" || Array.isArray(o)) return { ok: false, error: "grok-login" };
  for (const k of Object.keys(o)) {
    const key = o[k] && typeof o[k].key === "string" ? o[k].key.trim() : "";
    if (key) return { ok: true, key: key };
  }
  return { ok: false, error: "grok-login" };
}
```

```js
// api.anthropic.com/api/oauth/usage -> normalized snapshot (AD-004).
// Request contract (QML side): GET, headers Authorization: Bearer <token>,
// anthropic-beta: oauth-2025-04-20, User-Agent: claude-cli/<ver> (external, cli).

var API_URL = "https://api.anthropic.com/api/oauth/usage";
var BETA = "oauth-2025-04-20";
var HOUR = 3600000;
var WINDOW_MS = { session: 5 * HOUR, weekly: 168 * HOUR, "weekly-opus": 168 * HOUR, "weekly-sonnet": 168 * HOUR };
var KEYS = [["five_hour", "session"], ["seven_day", "weekly"], ["seven_day_opus", "weekly-opus"], ["seven_day_sonnet", "weekly-sonnet"]];

function windowFrom(raw, id) {
  if (!raw || typeof raw.utilization !== "number") return null;
  const end = raw.resets_at ? Date.parse(raw.resets_at) : NaN;
  return {
    id: id,
    percent: raw.utilization,
    start: isNaN(end) ? null : end - WINDOW_MS[id],
    end: isNaN(end) ? null : end
  };
}

function extraFrom(raw) {
  if (!raw || typeof raw.monthly_limit !== "number") return null;
  const scale = Math.pow(10, Number(raw.decimal_places) || 0);
  return {
    enabled: !!raw.is_enabled,
    used: Number(raw.used_credits || 0) / scale,
    limit: raw.monthly_limit / scale,
    percent: Number(raw.utilization || 0),
    currency: String(raw.currency || "")
  };
}

function parseUsage(text, plan, nowMs) {
  let o;
  try {
    o = JSON.parse(String(text || ""));
  } catch (e) {
    return { ok: false, error: "parse" };
  }
  if (!o || typeof o !== "object") return { ok: false, error: "parse" };
  const windows = KEYS.map((k) => windowFrom(o[k[0]], k[1])).filter((w) => w);
  if (windows.length === 0) return { ok: false, error: "parse" };
  return {
    ok: true,
    snapshot: { provider: "claude", plan: plan || "", windows: windows, extra: extraFrom(o.extra_usage), products: [], fetchedAt: nowMs }
  };
}
```

```js
// cli-chat-proxy.grok.com/v1/billing?format=credits -> normalized snapshot (AD-004).
// Request contract (QML side): GET, headers Authorization: Bearer <key>,
// X-XAI-Token-Auth: xai-grok-cli. Without format=credits the server answers a
// legacy shape with no percent (refused as "grok-format").

var API_URL = "https://cli-chat-proxy.grok.com/v1/billing?format=credits";
var TOKEN_AUTH = "xai-grok-cli";

function time(iso) {
  const t = iso ? Date.parse(iso) : NaN;
  return isNaN(t) ? null : t;
}

function parseBilling(text, nowMs) {
  let o;
  try {
    o = JSON.parse(String(text || ""));
  } catch (e) {
    return { ok: false, error: "parse" };
  }
  const c = o && o.config;
  if (!c || typeof c !== "object") return { ok: false, error: "parse" };
  if (typeof c.creditUsagePercent !== "number") return { ok: false, error: "grok-format" };
  const p = c.currentPeriod || {};
  const window = {
    id: "period",
    periodType: /MONTHLY/.test(String(p.type || "")) ? "monthly" : "weekly",
    percent: c.creditUsagePercent,
    start: time(p.start),
    end: time(p.end)
  };
  const products = (Array.isArray(c.productUsage) ? c.productUsage : [])
    .filter((x) => x && typeof x.usagePercent === "number")
    .map((x) => ({ name: String(x.product || ""), percent: x.usagePercent }));
  return { ok: true, snapshot: { provider: "grok", plan: "", windows: [window], extra: null, products: products, fetchedAt: nowMs } };
```

**Implement** `utils/pace.js`
```js
.pragma library

var CRITICAL = 90;
var AHEAD_POINTS = 10;
var PRIMARY = ["session", "period"];

// { elapsed, delta } in percent / points; null when the window has no bounds.
function pace(w, nowMs) {
  if (!w || w.start === null || w.end === null || !(w.end > w.start)) return { elapsed: null, delta: null };
  const elapsed = Math.max(0, Math.min(100, ((nowMs - w.start) / (w.end - w.start)) * 100));
  return { elapsed: Math.round(elapsed), delta: Math.round(w.percent - elapsed) };
}

function status(w, nowMs) {
  if (!w) return "ok";
  if (w.percent >= CRITICAL) return "critical";
  const p = pace(w, nowMs);
  return p.delta !== null && p.delta > AHEAD_POINTS ? "ahead" : "ok";
}

// The window the bar shows: Claude's 5 h session, Grok's period, else the first.
function primaryWindow(snapshot) {
  if (!snapshot || !snapshot.windows || snapshot.windows.length === 0) return null;
  for (const id of PRIMARY) {
    const w = snapshot.windows.find((x) => x.id === id);
    if (w) return w;
  }
  return snapshot.windows[0];
}
```

**Implement** `utils/poll.js`
```js
.pragma library

// Per-provider request state. Plain data, replaced (never mutated) on every step.
// error: "" | "network" | "http" | "auth" | "rate-limited" | "parse" | "grok-format"
//        | "claude-login" | "claude-expired" | "grok-login"

var BACKOFF_MS = 5 * 60000;

function createState() {
  return { inFlight: false, snapshot: null, stale: false, error: "", code: 0, retryAt: 0 };
}

function canRequest(s, nowMs) {
  return !s.inFlight && nowMs >= s.retryAt;
}

function begin(s, nowMs) {
  return Object.assign({}, s, { inFlight: true });
}

function succeed(s, snapshot) {
  return Object.assign({}, s, { inFlight: false, snapshot: snapshot, stale: false, error: "", code: 0, retryAt: 0 });
}

// kind: "network" | "parse" | "grok-format" | credential errors | "http" (with code)
function fail(s, kind, code, nowMs) {
  let error = kind;
  let retryAt = s.retryAt;
  if (kind === "http") {
    if (code === 401 || code === 403) error = "auth";
    else if (code === 429) {
      error = "rate-limited";
      retryAt = nowMs + BACKOFF_MS;
    }
  }
  return Object.assign({}, s, { inFlight: false, stale: s.snapshot !== null, error: error, code: kind === "http" ? code : 0, retryAt: retryAt });
}
```

```js
.pragma library

function countdown(endMs, nowMs) {
  if (endMs === null || endMs === undefined) return "";
  const min = Math.max(0, Math.floor((endMs - nowMs) / 60000));
  const d = Math.floor(min / 1440);
  const h = Math.floor((min % 1440) / 60);
  const m = min % 60;
  if (d > 0) return d + "d" + h + "h";
  if (h > 0) return h + "h" + m + "m";
  return m + "m";
}

function percentText(p) {
  return typeof p === "number" ? Math.round(p) + "%" : "—";
}

// "47% · 1h12m" for the primary window; "—" without data.
function barText(w, nowMs) {
  if (!w) return "—";
  const reset = countdown(w.end, nowMs);
  return reset ? percentText(w.percent) + " · " + reset : percentText(w.percent);
}

function minutesAgo(fetchedAt, nowMs) {
  return Math.max(0, Math.floor((nowMs - fetchedAt) / 60000));
}
```

**Implement** `utils/notify.js`
```js
.pragma library

// armed: { "<provider>|<window>": <end ms of the window that already fired> }.
// A window fires once per reset period; dropping below the threshold re-arms it.
function crossings(snapshot, armed, threshold) {
  const next = Object.assign({}, armed || {});
  const fire = [];
  if (!snapshot || !(threshold > 0)) return { fire: fire, armed: next };
  for (const w of snapshot.windows || []) {
    const key = snapshot.provider + "|" + w.id;
    if (w.percent < threshold) {
      delete next[key];
      continue;
    }
    if (next[key] === w.end) continue;
    next[key] = w.end;
    fire.push({ provider: snapshot.provider, id: w.id, percent: w.percent, end: w.end });
  }
  return { fire: fire, armed: next };
}
```
