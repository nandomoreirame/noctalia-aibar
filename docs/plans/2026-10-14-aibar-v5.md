## Global rules for every task

- TDD: write the Luau test, run it and see it fail, implement, run it green, then the full
  suite: `./run-tests` (read the `N passed, M failed` line and the `claude-swap:` line; never
  trust a piped exit code).
- Single file: `./run-tests tests/<name>.test.luau` (added in Task 1).
- `lib/*.luau` is pure: no `noctalia`, `barWidget`, `panel`, `ui` globals. Modules return one
  table. Lib modules require each other with `require("./x.luau")` (host form). Tests load
  them with `modules.load("lib/x.luau")`, never with the Luau CLI `require` directly.
- No `--!strict` in new files (the CLI does not type-check; keep files honest).
- Luau pitfalls: `end` is a keyword (use `startMs`/`endMs`); `table.insert(t, string.gsub(...))`
  passes two values (wrap the gsub in parentheses); storing `nil` deletes a key; `""` is truthy.
- Never write a real token, email or a `/home/<user>` path. Fixtures use `/data/u`, tests use
  `~/...` paths and fake tokens `SECRET-CLAUDE` / `SECRET-GROK`.
- Tokens only in the `headers` of `noctalia.http`. Never in `noctalia.state`, `noctalia.log`,
  `noctalia.writeFile`, `noctalia.notify`.
- English code and comments. UI strings via `noctalia.tr` with both translation files.
- No git commands inside tasks except the per-task `/git commit`.

```lua
  provider = "claude" | "grok",
```

```lua
-- Loads plugin modules for tests the way the Noctalia host does.
--
-- The host resolves require("./x.luau") relative to the requiring file and needs
-- the .luau suffix; the Luau CLI rejects that suffix. run-tests wraps every
-- source (lib/*.luau, root *.luau, fixtures, JSON, plugin.toml) into
-- tests/.generated/<sanitized>.src.luau returning the text, and this module
-- compiles it with a host-style require injected into the chunk env.

local Modules = {}

local cache: { [string]: any } = {}

-- "lib/pace.luau" -> "../.generated/lib_pace.src"; "translations/pt-BR.json" -> "../.generated/translations_pt-BR_json.src"
function Modules.generated(path: string): string
	local base = string.gsub(path, "^%./", "")
	base = string.gsub(base, "%.luau$", "")
	base = string.gsub(base, "[/\\.]", "_")
	return "../.generated/" .. base .. ".src"
end

local function dirOf(path: string): string
	return string.match(path, "^(.*)/[^/]*$") or ""
end

local function join(dir: string, rel: string): string
	local parts = {}
	for part in string.gmatch(dir, "[^/]+") do
		table.insert(parts, part)
	end
	for part in string.gmatch(rel, "[^/]+") do
		if part == ".." then
			table.remove(parts)
		elseif part ~= "." then
			table.insert(parts, part)
		end
	end
	return table.concat(parts, "/")
end

function Modules.text(path: string): string
	local ok, source = pcall(function()
		return require(Modules.generated(path))
	end)
	if not ok or typeof(source) ~= "string" then
		error(string.format("modules: no source for %s (run ./run-tests): %s", path, tostring(source)), 2)
	end
	return source
end

function Modules.requirer(fromPath: string): (string) -> any
	return function(rel: string): any
		if typeof(rel) ~= "string" or not string.match(rel, "^%.%.?/") or not string.match(rel, "%.luau$") then
			error("require path must be relative and end in .luau", 2)
		end
		return Modules.load(join(dirOf(fromPath), rel))
	end
end

function Modules.load(path: string): any
	if cache[path] ~= nil then
		return cache[path]
	end
	local chunk, err = loadstring(Modules.text(path), "@" .. path)
	if chunk == nil then
		error(string.format("modules: compile error in %s: %s", path, tostring(err)), 2)
	end
	setfenv(chunk, setmetatable({ require = Modules.requirer(path) }, { __index = _G }))
	local value = chunk()
	if value == nil then
		error(string.format("modules: %s returned nil", path), 2)
	end
	cache[path] = value
	return value
end

return Modules
```

```bash
#!/usr/bin/env bash
# scripts/claude-swap in a throwaway HOME with a fake pgrep (exit 1 = no claude running).
set -eo pipefail

SCRIPT="$(cd "$(dirname "${BASH_SOURCE[0]}")/.." && pwd)/scripts/claude-swap"
passed=0
failed=0

cred() { printf '{"claudeAiOauth":{"accessToken":"%s","refreshToken":"r-%s","expiresAt":1}}' "$1" "$1"; }
account() { printf '{"accountUuid":"%s","emailAddress":"%s@x.dev"}' "$1" "$1"; }

stub_pgrep() {
  printf '#!/bin/sh\nexit %s\n' "$1" >"$H/bin/pgrep"
  chmod +x "$H/bin/pgrep"
}

setup() {
  H="$(mktemp -d)"
  mkdir -p "$H/bin" "$H/.claude" "$H/.claude-freeze"
  (umask 077; cred A >"$H/.claude/.credentials.json"; cred B >"$H/.claude-freeze/.credentials.json")
  printf '{"oauthAccount":%s,"mcpServers":{"m":1}}' "$(account a)" >"$H/.claude.json"
  printf '{"oauthAccount":%s}' "$(account b)" >"$H/.claude-freeze/.claude.json"
  stub_pgrep 1
}

run_swap() {
  env -i HOME="$H" PATH="$H/bin:/usr/bin:/bin" timeout 20 bash "$SCRIPT" "$@" >"$H/out" 2>"$H/err"
}

field() { jq -c "$2" "$H/$1"; }
mode() { stat -c %a "$H/$1"; }

check() {
  local name="$1"
  shift
  if "$@"; then
    passed=$((passed + 1))
    echo "  ok $name"
  else
    failed=$((failed + 1))
    echo "  FAIL $name"
  fi
  [[ -n "$H" ]] && rm -rf "$H"
  H=""
}

t_swap() {
  setup
  run_swap || return 1
  [[ $(field .claude/.credentials.json .claudeAiOauth.accessToken) == '"B"' ]] &&
    [[ $(field .claude-freeze/.credentials.json .claudeAiOauth.accessToken) == '"A"' ]] &&
    [[ $(field .claude.json .) == '{"oauthAccount":{"accountUuid":"b","emailAddress":"b@x.dev"},"mcpServers":{"m":1}}' ]] &&
    [[ $(field .claude-freeze/.claude.json .) == '{"oauthAccount":{"accountUuid":"a","emailAddress":"a@x.dev"}}' ]] &&
    grep -q 'b@x.dev' "$H/out" && ! grep -q 'r-A' "$H/out"
}

t_swap_back() {
  setup
  run_swap && run_swap || return 1
  [[ $(field .claude/.credentials.json .claudeAiOauth.accessToken) == '"A"' ]] &&
    [[ $(field .claude.json .oauthAccount.accountUuid) == '"a"' ]]
}

t_private() {
  setup
  run_swap || return 1
  [[ $(mode .claude/.credentials.json) == 600 ]] &&
    [[ $(mode .claude-freeze/.credentials.json) == 600 ]] &&
    [[ $(mode .claude-freeze/.claude.json) == 600 ]]
}

t_running() {
  setup
  stub_pgrep 0
  run_swap && return 1
  grep -q claude "$H/err" &&
    [[ $(field .claude/.credentials.json .claudeAiOauth.accessToken) == '"A"' ]] &&
    [[ $(field .claude.json .oauthAccount.accountUuid) == '"a"' ]]
}

t_incomplete() {
  setup
  echo '{}' >"$H/.claude-freeze/.claude.json"
  run_swap && return 1
  [[ $(field .claude/.credentials.json .claudeAiOauth.accessToken) == '"A"' ]]
}

t_status() {
  setup
  run_swap status || return 1
  grep -q 'a@x.dev' "$H/out" && grep -q 'b@x.dev' "$H/out" &&
    [[ $(field .claude/.credentials.json .claudeAiOauth.accessToken) == '"A"' ]]
}

t_init() {
  H="$(mktemp -d)"
  mkdir -p "$H/bin" "$H/.claude"
  (umask 077; cred A >"$H/.claude/.credentials.json")
  printf '{"oauthAccount":%s,"x":1}' "$(account a)" >"$H/.claude.json"
  stub_pgrep 1
  run_swap init || return 1
  [[ $(field .claude-freeze/.credentials.json .claudeAiOauth.accessToken) == '"A"' ]] &&
    [[ $(field .claude-freeze/.claude.json .) == '{"oauthAccount":{"accountUuid":"a","emailAddress":"a@x.dev"}}' ]] &&
    [[ ! -e "$H/.claude/.credentials.json" ]] &&
    [[ $(field .claude.json .) == '{"x":1}' ]] &&
    [[ $(mode .claude-freeze) == 700 ]]
}

echo "claude-swap"
check "swaps logins and accounts, keeping the rest of .claude.json" t_swap
check "swaps back to the original state" t_swap_back
check "keeps the credential files private" t_private
check "refuses while claude is running and changes nothing" t_running
check "refuses when the frozen account is incomplete" t_incomplete
check "prints both accounts with status and changes nothing" t_status
check "freezes the current login into an empty freeze dir with init" t_init
echo "claude-swap: $passed passed, $failed failed"
((failed == 0))
```

```lua
-- ISO 8601 timestamps (as the usage APIs send them) to UTC epoch milliseconds.
-- No os.time: it converts through the local timezone.

local Time = {}

-- Days since 1970-01-01 for a proleptic Gregorian date (Howard Hinnant).
local function daysFromCivil(y: number, m: number, d: number): number
	if m <= 2 then
		y -= 1
	end
	local era = (if y >= 0 then y else y - 399) // 400
	local yoe = y - era * 400
	local mp = (m + 9) % 12
	local doy = (153 * mp + 2) // 5 + d - 1
	local doe = yoe * 365 + yoe // 4 - yoe // 100 + doy
	return era * 146097 + doe - 719468
end

function Time.parseIso(text: any): number?
	if typeof(text) ~= "string" then
		return nil
	end
	local y, mo, d, h, mi, s, rest = string.match(text, "^(%d%d%d%d)%-(%d%d)%-(%d%d)[Tt ](%d%d):(%d%d):(%d%d)(.*)$")
	if y == nil then
		return nil
	end
	local frac, zone = string.match(rest, "^%.(%d+)(.*)$")
	if frac == nil then
		frac, zone = "", rest
	end
	local offsetMin = 0
	if zone ~= "Z" and zone ~= "z" then
		local sign, oh, om = string.match(zone, "^([+-])(%d%d):?(%d%d)$")
		if sign == nil then
			return nil
		end
		offsetMin = (tonumber(oh) * 60 + tonumber(om)) * (if sign == "-" then -1 else 1)
	end
	local Y, M, D = tonumber(y), tonumber(mo), tonumber(d)
	local H, MI, S = tonumber(h), tonumber(mi), tonumber(s)
	if M < 1 or M > 12 or D < 1 or D > 31 or H > 23 or MI > 59 or S > 60 then
		return nil
	end
	local ms = tonumber(string.sub(frac .. "000", 1, 3))
	local seconds = daysFromCivil(Y, M, D) * 86400 + H * 3600 + MI * 60 + S - offsetMin * 60
	return seconds * 1000 + ms
end

return Time
```

```lua
-- Reads the CLIs' own credential data (decoded JSON in, plain result out).
-- Never logs, copies or writes a token; the service puts it straight into a
-- request header.

local Creds = {}

Creds.EXPIRY_MARGIN_MS = 60000

-- ~/.claude/.credentials.json -> { ok, token, plan } | { ok = false, error }
function Creds.claudeToken(data: any, nowMs: number)
	local oauth = if typeof(data) == "table" then data.claudeAiOauth else nil
	if typeof(oauth) ~= "table" or typeof(oauth.accessToken) ~= "string" or oauth.accessToken == "" then
		return { ok = false, error = "claude-login" }
	end
	local expiresAt = tonumber(oauth.expiresAt)
	if expiresAt == nil or not (expiresAt > nowMs + Creds.EXPIRY_MARGIN_MS) then
		return { ok = false, error = "claude-expired" }
	end
	return { ok = true, token = oauth.accessToken, plan = tostring(oauth.subscriptionType or "") }
end

-- ~/.grok/auth.json -> { ok, key } | { ok = false, error }. Decoded tables lose
-- the JSON key order, so issuers are tried in alphabetical order.
function Creds.grokKey(data: any)
	if typeof(data) ~= "table" or data[1] ~= nil then
		return { ok = false, error = "grok-login" }
	end
	local issuers = {}
	for issuer in data do
		if typeof(issuer) == "string" then
			table.insert(issuers, issuer)
		end
	end
	table.sort(issuers)
	for _, issuer in issuers do
		local entry = data[issuer]
		local key = if typeof(entry) == "table" and typeof(entry.key) == "string" then string.match(entry.key, "^%s*(.-)%s*$") else ""
		if key ~= "" then
			return { ok = true, key = key }
		end
	end
	return { ok = false, error = "grok-login" }
end

return Creds
```

```lua
-- Usage versus elapsed time of a window, status colors and window selection.

local Pace = {}

Pace.CRITICAL = 90
Pace.AHEAD_POINTS = 10
local PRIMARY = { "session", "period" }

local function round(x: number): number
	return math.floor(x + 0.5)
end

-- { elapsed, delta } in percent / points; {} when the window has no bounds.
function Pace.pace(w: any, nowMs: number)
	if not w or w.startMs == nil or w.endMs == nil or not (w.endMs > w.startMs) then
		return {}
	end
	local elapsed = math.clamp((nowMs - w.startMs) / (w.endMs - w.startMs) * 100, 0, 100)
	return { elapsed = round(elapsed), delta = round(w.percent - elapsed) }
end

function Pace.status(w: any, nowMs: number): string
	if not w then
		return "ok"
	end
	if w.percent >= Pace.CRITICAL then
		return "critical"
	end
	local p = Pace.pace(w, nowMs)
	if p.delta ~= nil and p.delta > Pace.AHEAD_POINTS then
		return "ahead"
	end
	return "ok"
end

-- The window the bar shows: Claude's 5 h session, Grok's period, else the first.
function Pace.primaryWindow(snapshot: any): any
	if not snapshot or not snapshot.windows or #snapshot.windows == 0 then
		return nil
	end
	for _, id in PRIMARY do
		for _, w in snapshot.windows do
			if w.id == id then
				return w
			end
		end
	end
	return snapshot.windows[1]
end

-- A cached snapshot can outlive its windows: one whose reset passed is shown as
-- 0% without bounds. Same table when nothing reset.
function Pace.project(snapshot: any, nowMs: number): any
	if not snapshot or not snapshot.windows then
		return snapshot
	end
	local changed = false
	local windows = {}
	for i, w in snapshot.windows do
		if w.endMs == nil or w.endMs > nowMs then
			windows[i] = w
		else
			changed = true
			local copy = table.clone(w)
			copy.percent = 0
			copy.startMs = nil
			copy.endMs = nil
			windows[i] = copy
		end
	end
	if not changed then
		return snapshot
	end
	local out = table.clone(snapshot)
	out.windows = windows
	return out
end

-- The window at 100% that unblocks last, or nil.
function Pace.exhausted(snapshot: any, nowMs: number): any
	if not snapshot or not snapshot.windows then
		return nil
	end
	local found = nil
	for _, w in snapshot.windows do
		if typeof(w.percent) == "number" and w.percent >= 100 and w.endMs ~= nil and w.endMs > nowMs then
			if found == nil or w.endMs > found.endMs then
				found = w
			end
		end
	end
	return found
end

return Pace
```

```lua
-- Text pieces shared by the bar, tooltip, panel and notifications. Returns
-- i18n keys and vars, never translated text.

local Format = {}

function Format.countdown(endMs: number?, nowMs: number): string
	if endMs == nil then
		return ""
	end
	local min = math.max(0, math.floor((endMs - nowMs) / 60000))
	local d = min // 1440
	local h = (min % 1440) // 60
	local m = min % 60
	if d > 0 then
		return d .. "d" .. h .. "h"
	end
	if h > 0 then
		return h .. "h" .. m .. "m"
	end
	return m .. "m"
end

function Format.percentText(p: any): string
	if typeof(p) ~= "number" then
		return "—"
	end
	return math.floor(p + 0.5) .. "%"
end

-- "47% · 1h12m" for the primary window; "—" without data.
function Format.barText(w: any, nowMs: number): string
	if not w then
		return "—"
	end
	local reset = Format.countdown(w.endMs, nowMs)
	if reset == "" then
		return Format.percentText(w.percent)
	end
	return Format.percentText(w.percent) .. " · " .. reset
end

function Format.minutesAgo(fetchedAt: number, nowMs: number): number
	return math.max(0, math.floor((nowMs - fetchedAt) / 60000))
end

-- "5m", "2h5m", "3d0h" since fetchedAt.
function Format.ago(fetchedAt: number, nowMs: number): string
	return Format.countdown(nowMs, fetchedAt)
end

function Format.windowKey(w: any): string
	if w.id == "period" then
		return "window.period-" .. (if w.periodType == "monthly" then "monthly" else "weekly")
	end
	return "window." .. w.id
end

local LOGIN_ERRORS = { auth = true, ["grok-login"] = true, ["claude-login"] = true, ["claude-expired"] = true }

-- { key, vars? } for a source's request error; nil when there is none. quiet
-- hides login errors: the CLIs renew their token only when used, so an expired
-- one is expected while a plan is blocked or an account is frozen.
function Format.errorMessage(usage: any, provider: string, nowMs: number, quiet: boolean?): any
	if not usage or usage.error == nil or usage.error == "" then
		return nil
	end
	local e = usage.error
	if quiet and LOGIN_ERRORS[e] then
		return nil
	end
	if e == "auth" then
		return { key = if provider == "grok" then "error.grok-login" else "error.claude-expired" }
	end
	if e == "http" then
		return { key = "error.http", vars = { code = usage.code } }
	end
	if e == "rate-limited" then
		return { key = "error.rate-limited", vars = { n = math.max(0, math.ceil(((usage.retryAt or 0) - nowMs) / 60000)) } }
	end
	return { key = "error." .. e }
end

function Format.limitMessage(w: any, nowMs: number): any
	if not w then
		return nil
	end
	return { key = "panel.limit-reached", vars = { t = Format.countdown(w.endMs, nowMs) } }
end

-- "33,50 / 90,00 BRL"; the decimal separator comes from the UI language.
function Format.moneyText(extra: any, decimalSep: string?): string
	if not extra then
		return ""
	end
	local function amount(n: number): string
		local text = string.format("%.2f", n)
		if decimalSep == "," then
			text = (string.gsub(text, "%.", ","))
		end
		return text
	end
	local currency = if extra.currency and extra.currency ~= "" then " " .. extra.currency else ""
	return amount(extra.used) .. " / " .. amount(extra.limit) .. currency
end
```

```lua
-- Per-source request state. Plain data, replaced (never mutated) on every step.
-- error: "" | "network" | "http" | "auth" | "rate-limited" | "parse" | "grok-format"
--        | "claude-login" | "claude-expired" | "grok-login"

local Poll = {}

Poll.BACKOFF_MS = 5 * 60000
-- noctalia.http times out at 30 s; this guard also covers a callback lost to a
-- reload or suspend.
Poll.REQUEST_TIMEOUT_MS = 30000

local function with(s: any, fields: any): any
	local out = table.clone(s)
	for k, v in fields do
		out[k] = v
	end
	return out
end

function Poll.createState()
	return { inFlight = false, startedAt = 0, stale = false, error = "", code = 0, retryAt = 0 }
end

function Poll.canRequest(s: any, nowMs: number): boolean
	local free = not s.inFlight or nowMs - s.startedAt >= Poll.REQUEST_TIMEOUT_MS
	return free and nowMs >= s.retryAt
end

function Poll.begin(s: any, nowMs: number): any
	return with(s, { inFlight = true, startedAt = nowMs })
end

function Poll.succeed(s: any, snapshot: any): any
	return with(s, { inFlight = false, snapshot = snapshot, stale = false, error = "", code = 0, retryAt = 0 })
end

-- kind: "network" | "parse" | "grok-format" | credential errors | "http" (with code)
function Poll.fail(s: any, kind: string, code: number, nowMs: number): any
	local err = kind
	local retryAt = s.retryAt
	if kind == "http" then
		if code == 401 or code == 403 then
			err = "auth"
		elseif code == 429 then
			err = "rate-limited"
			retryAt = nowMs + Poll.BACKOFF_MS
		end
	end
	return with(s, {
		inFlight = false,
		stale = s.snapshot ~= nil,
		error = err,
		code = if kind == "http" then code else 0,
		retryAt = retryAt,
	})
end

-- Seeds an empty state with the snapshot cached on disk, stale until the first
-- answer. Live data always wins over the cache.
function Poll.restore(s: any, snapshot: any): any
	if s.snapshot ~= nil or typeof(snapshot) ~= "table" or typeof(snapshot.windows) ~= "table" then
		return s
	end
	return with(s, { snapshot = snapshot, stale = true })
end

return Poll
```

```lua
-- armed: { ["<source>|<window>"] = <endMs of the window that already fired, or -1> }.
-- A window fires once per reset period; dropping below the threshold re-arms it.

local Notify = {}

local NO_END = -1

function Notify.crossings(snapshot: any, armed: any, threshold: any, source: string?)
	local nextArmed = table.clone(armed or {})
	local fire = {}
	if not snapshot or typeof(threshold) ~= "number" or threshold <= 0 then
		return { fire = fire, armed = nextArmed }
	end
	for _, w in snapshot.windows or {} do
		local key = (source or snapshot.provider) .. "|" .. w.id
		local stamp = if w.endMs ~= nil then w.endMs else NO_END
		if w.percent < threshold then
			nextArmed[key] = nil
		elseif nextArmed[key] ~= stamp then
			nextArmed[key] = stamp
			table.insert(fire, { provider = snapshot.provider, id = w.id, periodType = w.periodType, percent = w.percent, endMs = w.endMs })
		end
	end
	return { fire = fire, armed = nextArmed }
end

return Notify
```

```lua
-- Usage sources: the active Claude account (~/.claude), frozen Claude accounts
-- (each a CLAUDE_CONFIG_DIR holding the login of an account not in use, see
-- scripts/claude-swap) and Grok. Keys: "claude", "claude:<slug>", "grok".
-- Paths stay in "~/..." form; noctalia.readFile expands "~".

local Accounts = {}

Accounts.MAX_EXTRA = 4
Accounts.MAX_NAME = 24
local DEFAULT_DIR = "~/.claude"

local function trim(s: string): string
	return (string.gsub(s, "^%s*(.-)%s*$", "%1"))
end

local function slug(name: string): string
	local s = string.gsub(string.lower(name), "[^a-z0-9]+", "-")
	return (string.gsub(s, "^%-+", ""):gsub("%-+$", ""))
end

local function validDir(dir: string): boolean
	if string.sub(dir, 1, 2) ~= "~/" and string.sub(dir, 1, 1) ~= "/" then
		return false
	end
	if string.find(dir, "%c") then
		return false
	end
	for part in string.gmatch(dir, "[^/]+") do
		if part == ".." then
			return false
		end
	end
	return true
end

-- { "Name=dir", ... } (or one "a=b, c=d" string) -> { { key, name, dir } }.
-- Invalid, duplicate and default-dir entries are dropped.
function Accounts.parseAccounts(value: any)
	local entries = {}
	local function split(text: string)
		for part in string.gmatch(text, "[^,\n]+") do
			table.insert(entries, part)
		end
	end
	if typeof(value) == "string" then
		split(value)
	elseif typeof(value) == "table" then
		for _, item in value do
			if typeof(item) == "string" then
				split(item)
			end
		end
	end
	local out = {}
	local seen = {}
	for _, entry in entries do
		local eq = string.find(entry, "=", 1, true)
		if eq then
			local name = trim(string.sub(entry, 1, eq - 1))
			local cut = utf8.offset(name, Accounts.MAX_NAME + 1)
			if cut then
				name = trim(string.sub(name, 1, cut - 1))
			end
			local dir = (string.gsub(trim(string.sub(entry, eq + 1)), "/+$", ""))
			local id = slug(name)
			if id ~= "" and validDir(dir) and dir ~= DEFAULT_DIR and not seen[id] then
				seen[id] = true
				table.insert(out, { key = "claude:" .. id, name = name, dir = dir })
				if #out == Accounts.MAX_EXTRA then
					break
				end
			end
		end
	end
	return out
end

function Accounts.sources(extra: any)
	local list = { { key = "claude", provider = "claude", name = "", dir = "", frozen = false } }
	for _, a in extra or {} do
		table.insert(list, { key = a.key, provider = "claude", name = a.name, dir = a.dir, frozen = true })
	end
	table.insert(list, { key = "grok", provider = "grok", name = "", dir = "", frozen = false })
	return list
end

function Accounts.credPath(source: any): string
	if source.provider == "grok" then
		return "~/.grok/auth.json"
	end
	return (if source.dir ~= "" then source.dir else DEFAULT_DIR) .. "/.credentials.json"
end

-- Claude Code keeps .claude.json in HOME by default, inside CLAUDE_CONFIG_DIR otherwise.
function Accounts.configPath(source: any): string
	if source.dir == "" then
		return "~/.claude.json"
	end
	return source.dir .. "/.claude.json"
end

-- .claude.json data -> { uuid, email } of the logged account, or nil. The file holds no token.
function Accounts.accountOf(data: any): any
	local a = if typeof(data) == "table" then data.oauthAccount else nil
	if typeof(a) ~= "table" or typeof(a.accountUuid) ~= "string" or a.accountUuid == "" then
		return nil
	end
	return { uuid = a.accountUuid, email = if typeof(a.emailAddress) == "string" then a.emailAddress else "" }
end

-- "adopt" the first id, "reset" after a swap, "none" otherwise. An unreadable
-- file (mid-write) keeps the last id.
function Accounts.accountChange(prev: string, nextId: string): string
	if nextId == nil or nextId == "" or nextId == prev then
		return "none"
	end
	return if prev ~= "" then "reset" else "adopt"
end

-- Snapshots are cached by account id, so a swap never shows one account's usage
-- under the other. Without an id, the per-provider entry is used.
function Accounts.cachedSnapshot(cache: any, key: string, uuid: string): any
	if typeof(cache) ~= "table" then
		return nil
	end
	if uuid ~= "" then
		local byAccount = if typeof(cache.byAccount) == "table" then cache.byAccount else {}
		return byAccount[uuid]
	end
	if key == "claude" or key == "grok" then
		return cache[key]
	end
	return nil
end

function Accounts.indexOfKey(list: any, key: string): number
	for i, s in list do
		if s.key == key then
			return i
		end
	end
	return 0
end

function Accounts.resolveKey(list: any, key: string): string
	if Accounts.indexOfKey(list, key) > 0 then
		return key
	end
	return list[1].key
end

function Accounts.nextKey(list: any, key: string): string
	local i = Accounts.indexOfKey(list, key)
	if i == 0 then
		return list[1].key
	end
	return list[i % #list + 1].key
end

return Accounts
```

```lua
-- Request contract (service): GET with Authorization: Bearer <token>,

local time = require("./time.luau")

local Claude = {}

Claude.API_URL = "https://api.anthropic.com/api/oauth/usage"
Claude.BETA = "oauth-2025-04-20"
local HOUR = 3600000
Claude.WINDOW_MS = { session = 5 * HOUR, weekly = 168 * HOUR, ["weekly-opus"] = 168 * HOUR, ["weekly-sonnet"] = 168 * HOUR }
local KEYS = { { "five_hour", "session" }, { "seven_day", "weekly" }, { "seven_day_opus", "weekly-opus" }, { "seven_day_sonnet", "weekly-sonnet" } }

local function windowFrom(raw: any, id: string): any
	if typeof(raw) ~= "table" or typeof(raw.utilization) ~= "number" then
		return nil
	end
	local endMs = time.parseIso(raw.resets_at)
	return {
		id = id,
		percent = raw.utilization,
		startMs = if endMs ~= nil then endMs - Claude.WINDOW_MS[id] else nil,
		endMs = endMs,
	}
end

local function extraFrom(raw: any): any
	if typeof(raw) ~= "table" or typeof(raw.monthly_limit) ~= "number" then
		return nil
	end
	local scale = 10 ^ (tonumber(raw.decimal_places) or 0)
	return {
		enabled = raw.is_enabled == true,
		used = (tonumber(raw.used_credits) or 0) / scale,
		limit = raw.monthly_limit / scale,
		percent = tonumber(raw.utilization) or 0,
		currency = tostring(raw.currency or ""),
	}
end

function Claude.parseUsage(data: any, plan: string?, nowMs: number)

		return { ok = false, error = "parse" }
	end

		if w then
			table.insert(windows, w)
		end
	end

		return { ok = false, error = "parse" }
	end

		ok = true,
		snapshot = { provider = "claude", plan = plan or "", windows = windows, extra = extraFrom(data.extra_usage), products = {}, fetchedAt = nowMs },
```

```lua
-- cli-chat-proxy.grok.com/v1/billing?format=credits (decoded) -> normalized snapshot.
-- Request contract (service): GET with Authorization: Bearer <key>,
-- X-XAI-Token-Auth: xai-grok-cli. Without format=credits the server answers a
-- legacy shape with no percent (refused as "grok-format").

local time = require("./time.luau")

local Grok = {}

Grok.API_URL = "https://cli-chat-proxy.grok.com/v1/billing?format=credits"
Grok.TOKEN_AUTH = "xai-grok-cli"

function Grok.parseBilling(data: any, nowMs: number)
	local c = if typeof(data) == "table" then data.config else nil
	if typeof(c) ~= "table" then
		return { ok = false, error = "parse" }
	end
	-- The server drops zero fields (protobuf JSON): a period without a percent
	-- is a fresh one at 0%. The legacy shape has neither.
	local hasPercent = typeof(c.creditUsagePercent) == "number"
	if not hasPercent and typeof(c.currentPeriod) ~= "table" then
		return { ok = false, error = "grok-format" }
	end
	local p = if typeof(c.currentPeriod) == "table" then c.currentPeriod else {}
	local window = {
		id = "period",
		periodType = if string.find(tostring(p.type or ""), "MONTHLY", 1, true) then "monthly" else "weekly",
		percent = if hasPercent then c.creditUsagePercent else 0,
		startMs = time.parseIso(p.start),
		endMs = time.parseIso(p["end"]),
	}
	local products = {}
	if typeof(c.productUsage) == "table" then
		for _, x in c.productUsage do
			if typeof(x) == "table" and typeof(x.usagePercent) == "number" then
				table.insert(products, { name = tostring(x.product or ""), percent = x.usagePercent })
			end
		end
	end
	return { ok = true, snapshot = { provider = "grok", plan = "", windows = { window }, products = products, fetchedAt = nowMs } }
end

return Grok
```

```lua
-- header). Copied across entry VMs, so it holds data only: never a token.

local accounts = require("./accounts.luau")
local pace = require("./pace.luau")

local View = {}

function View.build(sources: any, states: any, identities: any, barKey: string, nowMs: number)
	local list = {}
	for _, s in sources do
		local st = states[s.key] or {}
		local id = identities[s.key]
		table.insert(list, {
			key = s.key,
			provider = s.provider,
			name = s.name,
			frozen = s.frozen,
			email = if id then id.email else "",
			snapshot = pace.project(st.snapshot, nowMs),
			stale = st.stale == true,
			error = st.error or "",
			code = st.code or 0,
			retryAt = st.retryAt or 0,
			inFlight = st.inFlight == true,
		})
	end
	return { sources = list, barKey = barKey, now = nowMs }
end

-- The persisted middle-click choice wins at start while its source exists.
function View.initialKey(sources: any, persisted: any, setting: any): string
	if typeof(persisted) == "string" and accounts.indexOfKey(sources, persisted) > 0 then
		return persisted
	end
	return accounts.resolveKey(sources, if typeof(setting) == "string" then setting else "")
end
```

```lua
-- View -> display models for the bar widget and the panel. Pure: every string
-- comes from the tr(key, vars?) function the entry passes in.

local format = require("./format.luau")
local pace = require("./pace.luau")

local Present = {}

local FILL = { ok = "primary", ahead = "tertiary", critical = "error" }

local function glyphOf(provider: string): string
	return if provider == "grok" then "brand-x" else "asterisk"
end

local function fraction(p: any): number
	if typeof(p) ~= "number" then
		return 0
	end
	return math.clamp(p, 0, 100) / 100
end

function Present.find(view: any, key: string): any
	if not view or not view.sources then
		return nil
	end
	for _, s in view.sources do
		if s.key == key then
			return s
		end
	end
	return nil
end

-- "Claude Code", "Claude Code · Trabalho" or "Grok".
function Present.sourceName(source: any, tr: any): string
	local base = tr("provider." .. source.provider)
	if source.name ~= nil and source.name ~= "" then
		return base .. " · " .. source.name
	end
	return base
end

function Present.errorText(entry: any, nowMs: number, tr: any): string
	local quiet = entry.frozen or pace.exhausted(entry.snapshot, nowMs) ~= nil
	local msg = format.errorMessage(entry, entry.provider, nowMs, quiet)
	if not msg then
		return ""
	end
	return tr(msg.key, msg.vars)
end

local function claudeCount(view: any): number
	local n = 0
	for _, s in view.sources do
		if s.provider == "claude" then
			n += 1
		end
	end
	return n
end

function Present.bar(view: any, tr: any)
	local entry = if view then Present.find(view, view.barKey) else nil
	if not entry then
		return { glyph = "asterisk", label = "—", fraction = 0, fill = "primary", stale = false, tooltip = {} }
	end
	local now = view.now or 0
	local snap = entry.snapshot
	local rows = {}
	if claudeCount(view) > 1 then
		table.insert(rows, { Present.sourceName(entry, tr), "" })
	end
	for _, w in (if snap then snap.windows else {}) do
		local value = format.percentText(w.percent)
		local reset = format.countdown(w.endMs, now)
		if reset ~= "" then
			value ..= " · " .. tr("panel.resets", { t = reset })
		end
		table.insert(rows, { tr(format.windowKey(w)), value })
	end
	local limit = format.limitMessage(pace.exhausted(snap, now), now)
	if limit then
		table.insert(rows, { tr(limit.key, limit.vars), "" })
	end
	local err = Present.errorText(entry, now, tr)
	if err ~= "" then
		table.insert(rows, { err, "" })
	end
	local w = pace.primaryWindow(snap)
	return {
		glyph = glyphOf(entry.provider),
		label = format.barText(w, now),
		fraction = fraction(if w then w.percent else nil),
		fill = FILL[pace.status(w, now)],
		stale = entry.stale,
		tooltip = rows,
	}
end

local function detailLine(w: any, nowMs: number, tr: any): string
	local reset = format.countdown(w.endMs, nowMs)
	local resets = if reset ~= "" then tr("panel.resets", { t = reset }) else ""
	local p = pace.pace(w, nowMs)
	local paceText = ""
	if p.delta ~= nil then
		local points = math.abs(p.delta)
		if points <= 2 then
			paceText = tr("pace.on")
		elseif p.delta > 0 then
			paceText = tr("pace.ahead", { n = points })
		else
			paceText = tr("pace.behind", { n = points })
		end
	end
	if resets ~= "" and paceText ~= "" then
		return resets .. " · " .. paceText
	end
	return if resets ~= "" then resets else paceText
end

function Present.card(entry: any, nowMs: number, tr: any)
	local snap = entry.snapshot
	local card = {
		key = entry.key,
		glyph = glyphOf(entry.provider),
		title = tr("provider." .. entry.provider),
		plan = "",
		email = entry.email or "",
		updated = "",
		badge = nil,
		rows = {},
		extra = nil,
		products = {},
		limit = "",
		error = Present.errorText(entry, nowMs, tr),
		empty = "",
	}
	if entry.frozen then
		card.badge = { text = tr("panel.frozen"), color = "primary" }
	elseif entry.stale then
		card.badge = { text = tr("panel.stale"), color = "error" }
	end
	if not snap then
		if card.error == "" then
			card.empty = tr("panel.no-data")
		end
		return card
	end
	if entry.provider == "claude" and typeof(snap.plan) == "string" and snap.plan ~= "" then
		card.plan = string.upper(string.sub(snap.plan, 1, 1)) .. string.sub(snap.plan, 2)
	end
	-- A frozen account's last status is the point, so its age stays visible.
	if entry.frozen or not entry.stale then
		if format.minutesAgo(snap.fetchedAt, nowMs) == 0 then
			card.updated = tr("panel.updated-now")
		else
			card.updated = tr("panel.updated", { t = format.ago(snap.fetchedAt, nowMs) })
		end
	end
	for _, w in snap.windows or {} do
		table.insert(card.rows, {
			label = tr(format.windowKey(w)),
			percentText = format.percentText(w.percent),
			fraction = fraction(w.percent),
			fill = FILL[pace.status(w, nowMs)],
			detail = detailLine(w, nowMs, tr),
		})
	end
	if snap.extra then
		card.extra = {
			label = tr("window.extra"),
			percentText = format.percentText(snap.extra.percent),
			fraction = fraction(snap.extra.percent),
			fill = if snap.extra.percent >= pace.CRITICAL then "error" else "primary",
			detail = format.moneyText(snap.extra, tr("format.decimal")),
		}
	end
	if entry.provider == "grok" then
		for _, p in snap.products or {} do
			table.insert(card.products, { name = p.name, percentText = format.percentText(p.percent) })
		end
	end
	local limit = format.limitMessage(pace.exhausted(snap, nowMs), nowMs)
	if limit then
		card.limit = tr(limit.key, limit.vars)
	end
	return card
end

-- claudeTab: 1-based index of the Claude account shown in the shared card.
function Present.panel(view: any, claudeTab: number, tr: any)
	local model = { title = tr("panel.title"), refresh = tr("panel.refresh"), busy = false, tabs = {}, claude = nil, others = {} }
	if not view or not view.sources then
		return model
	end
	local now = view.now or 0
	local claudes = {}
	for _, s in view.sources do
		if s.inFlight then
			model.busy = true
		end
		if s.provider == "claude" then
			table.insert(claudes, s)
		else
			table.insert(model.others, Present.card(s, now, tr))
		end
	end
	if #claudes > 0 then
		local index = math.clamp(claudeTab, 1, #claudes)
		model.claude = Present.card(claudes[index], now, tr)
		if #claudes > 1 then
			for i, s in claudes do
				table.insert(model.tabs, { label = if s.frozen then s.name else tr("panel.active"), selected = i == index })
			end
		end
	end
	return model
end

return Present
```

```lua
-- AI Bar service: the only entry that touches credentials, HTTP and files.
-- Tokens live only in locals of pollSource and in the headers table handed to
-- noctalia.http. They never reach noctalia.state, logs, files or notifications.

local accounts = require("./lib/accounts.luau")
local claude = require("./lib/claude.luau")
local creds = require("./lib/creds.luau")
local format = require("./lib/format.luau")
local grok = require("./lib/grok.luau")
local notify = require("./lib/notify.luau")
local poll = require("./lib/poll.luau")
local present = require("./lib/present.luau")
local view = require("./lib/view.luau")

local TICK_MS = 30000

local sources = {}
local states = {}
local identities = {}
local mtimes = {}
local generation = {}
local armed = {}
local cache = { byAccount = {} }
local barKey = "claude"
local lastBarSetting = ""
local nextPollAt = 0

local function now(): number
	return noctalia.nowMs()
end

local function dataPath(name: string): string
	return noctalia.pluginDataDir() .. "/" .. name
end

local function decodeFile(path: string): any
	local text = noctalia.readFile(path)
	if text == nil then
		return nil
	end
	return (noctalia.json.decode(text))
end

local function tr(key: string, vars: any?): string
	if vars then
		return noctalia.tr(key, vars)
	end
	return noctalia.tr(key)
end

local function stateOf(key: string): any
	return states[key] or poll.createState()
end

local function uuidOf(key: string): string
	local id = identities[key]
	return if id then id.uuid else ""
end

local function publish()
	noctalia.state.set("view", view.build(sources, states, identities, barKey, now()))
end

local function saveSnapshot(key: string, snapshot: any)
	local uuid = uuidOf(key)
	if uuid ~= "" then
		cache.byAccount[uuid] = snapshot
	elseif key == "claude" or key == "grok" then
		cache[key] = snapshot
	else
		return
	end
	noctalia.writeFile(dataPath("snapshots.json"), noctalia.json.encode(cache))
end

local function fireNotifications(source: any, snapshot: any)
	local result = notify.crossings(snapshot, armed, tonumber(noctalia.getConfig("notify_threshold")) or 0, source.key)
	armed = result.armed
	for _, item in result.fire do
		noctalia.notify(
			tr("notify.title", { provider = present.sourceName(source, tr), window = tr(format.windowKey(item)), n = math.floor(item.percent + 0.5) }),
			tr("notify.body", { t = format.countdown(item.endMs, now()) })
		)
	end
end

local function applyResult(source: any, kind: string, code: number, snapshot: any)
	local key = source.key
	if snapshot then
		states[key] = poll.succeed(stateOf(key), snapshot)
		fireNotifications(source, snapshot)
		saveSnapshot(key, snapshot)
	else
		states[key] = poll.fail(stateOf(key), kind, code, now())
	end
	publish()
end

-- headers is a local list; nothing in it is logged. An answer is dropped when a
-- newer request replaced it or the source's account changed meanwhile.
local function request(source: any, url: string, headers: { string }, parse: (any) -> any)
	local key = source.key
	local id = (generation[key] or 0) + 1
	generation[key] = id
	local account = uuidOf(key)
	states[key] = poll.begin(stateOf(key), now())
	noctalia.http({ url = url, method = "GET", headers = headers }, function(res)
		if generation[key] ~= id or uuidOf(key) ~= account then
			return
		end
		local status = tonumber(res and res.status) or 0
		if status ~= 200 then
			noctalia.log("aibar " .. key .. " " .. status)
			applyResult(source, if status == 0 then "network" else "http", status, nil)
			return
		end
		local parsed = parse(noctalia.json.decode(res.body or ""))
		if not parsed.ok then
			applyResult(source, parsed.error, 0, nil)
			return
		end
		applyResult(source, "", 0, parsed.snapshot)
	end)
end

local function pollSource(source: any)
	local key = source.key
	local t = now()
	local current = stateOf(key)
	if not poll.canRequest(current, t) then
		return
	end
	if source.provider == "grok" then
		local cred = creds.grokKey(decodeFile(accounts.credPath(source)))
		if not cred.ok then
			states[key] = poll.fail(current, cred.error, 0, t)
			return
		end
		request(source, grok.API_URL, {
			"Authorization: Bearer " .. cred.key,
			"X-XAI-Token-Auth: " .. grok.TOKEN_AUTH,
			"Accept: application/json",
		}, function(data)
			return grok.parseBilling(data, now())
		end)
		return
	end
	local cred = creds.claudeToken(decodeFile(accounts.credPath(source)), t)
	if not cred.ok then
		states[key] = poll.fail(current, cred.error, 0, t)
		return
	end
	local plan = cred.plan
	request(source, claude.API_URL, {
		"Authorization: Bearer " .. cred.token,
		"anthropic-beta: " .. claude.BETA,
		"User-Agent: " .. tostring(noctalia.getConfig("claude_user_agent") or ""),
		"Accept: application/json",
	}, function(data)
		return claude.parseUsage(data, plan, now())
	end)
end

local function pollAll()
	local minutes = math.max(2, tonumber(noctalia.getConfig("poll_minutes")) or 5)
	nextPollAt = now() + minutes * 60000
	for _, s in sources do
		pollSource(s)
	end
	publish()
end

-- A source's .claude.json was (re)read: claude-swap moves logins between
-- accounts, so the account behind a key can change.
local function syncIdentity(source: any)
	local key = source.key
	local info = accounts.accountOf(decodeFile(accounts.configPath(source)))
	local change = accounts.accountChange(uuidOf(key), if info then info.uuid else "")
	if change == "none" then
		return
	end
	identities[key] = info
	local current = stateOf(key)
	local cached = accounts.cachedSnapshot(cache, key, info.uuid)
	if change == "adopt" then
		if current.snapshot ~= nil and not current.stale then
			saveSnapshot(key, current.snapshot)
		elseif cached ~= nil then
			local nextState = table.clone(current)
			nextState.snapshot = cached
			nextState.stale = true
			states[key] = nextState
		end
		return
	end
	generation[key] = (generation[key] or 0) + 1
	states[key] = poll.restore(poll.createState(), cached)
	pollSource(source)
end

local function loadCache()
	local data = decodeFile(dataPath("snapshots.json"))
	if typeof(data) == "table" then
		cache = data
	end
	if typeof(cache.byAccount) ~= "table" then
		cache.byAccount = {}
	end
end

local function configure()
	sources = accounts.sources(accounts.parseAccounts(noctalia.getConfig("claude_accounts")))
	for _, s in sources do
		if s.provider == "claude" then
			syncIdentity(s)
		end
		if states[s.key] == nil then
			states[s.key] = poll.restore(poll.createState(), accounts.cachedSnapshot(cache, s.key, uuidOf(s.key)))
		end
	end
end

local function start()
	noctalia.setUpdateInterval(TICK_MS)
	loadCache()
	configure()
	lastBarSetting = tostring(noctalia.getConfig("bar_provider") or "")
	local selection = decodeFile(dataPath("selection.json"))
	barKey = view.initialKey(sources, if typeof(selection) == "table" then selection.key else nil, lastBarSetting)
	pollAll()
end

function update()
	if now() >= nextPollAt then
		pollAll()
	else
		publish()
	end
end

function onIpc(event: string, _payload: any)
	if event == "refresh" then
		for _, s in sources do
			pollSource(s)
		end
		publish()
	end
end

start()
```

```lua
-- AI Bar panel: one card per provider; Claude accounts share one card with a
-- tab row. Renders only from the service's `view`.

local present = require("./lib/present.luau")

local isOpen = false
local claudeTab = 1
local current = nil

local function tr(key: string, vars: any?): string
	if vars then
		return noctalia.tr(key, vars)
	end
	return noctalia.tr(key)
end

local function usageRow(row: any): any
	local children = {
		ui.row({ align = "center", gap = 8 }, {
			ui.label({ text = row.label, fontSize = 12, flexGrow = 1 }),
			ui.label({ text = row.percentText, fontSize = 12, fontWeight = "bold" }),
		}),
		ui.progress({ progress = row.fraction, fill = row.fill, track = "on_surface/0.15", height = 6, radius = 3 }),
	}
	if row.detail ~= "" then
		table.insert(children, ui.label({ text = row.detail, fontSize = 11, color = "on_surface_variant" }))
	end
	return ui.column({ gap = 4 }, children)
end

local function note(text: string, color: string): any
	return ui.label({ text = text, fontSize = 11, color = color })
end

local function card(c: any, tabs: any): any
	local header = {
		ui.glyph({ name = c.glyph, size = 16 }),
		ui.label({ text = c.title, fontSize = 14, fontWeight = "bold" }),
	}
	if c.plan ~= "" then
		table.insert(header, ui.label({ text = c.plan, fontSize = 12, color = "on_surface_variant" }))
	end
	table.insert(header, ui.spacer({ flexGrow = 1 }))
	if c.updated ~= "" then
		table.insert(header, ui.label({ text = c.updated, fontSize = 10, color = "on_surface_variant" }))
	end
	if c.badge then
		table.insert(header, ui.box({ fill = c.badge.color .. "/0.18", radius = 8 }, {
			ui.label({ text = c.badge.text, fontSize = 10, color = c.badge.color }),
		}))
	end
	local body = { ui.row({ align = "center", gap = 8 }, header) }
	if c.email ~= "" or #tabs > 0 then
		local line = { ui.label({ text = c.email, fontSize = 10, color = "on_surface_variant", flexGrow = 1 }) }
		for i, tab in tabs do
			table.insert(line, ui.button({
				key = "tab-" .. i,
				text = tab.label,
				variant = "ghost",
				controlSize = "sm",
				selected = tab.selected,
				onClick = function()
					claudeTab = i
					render()
				end,
			}))
		end
		table.insert(body, ui.row({ align = "center", gap = 4 }, line))
	end
	if c.empty ~= "" then
		table.insert(body, note(c.empty, "on_surface_variant"))
	end
	for _, row in c.rows do
		table.insert(body, usageRow(row))
	end
	if c.extra then
		table.insert(body, usageRow(c.extra))
	end
	for _, p in c.products do
		table.insert(body, ui.row({ gap = 8 }, {
			ui.label({ text = p.name, fontSize = 11, color = "on_surface_variant", flexGrow = 1 }),
			ui.label({ text = p.percentText, fontSize = 11, color = "on_surface_variant" }),
		}))
	end
	if c.limit ~= "" then
		table.insert(body, note(c.limit, "error"))
	end
	if c.error ~= "" then
		table.insert(body, note(c.error, "error"))
	end
	return ui.column({ key = "card-" .. c.key, gap = 8, padding = 12, radius = 12, fill = "surface_variant" }, body)
end

function render()
	if not isOpen then
		return
	end
	local m = present.panel(current, claudeTab, tr)
	local cards = {}
	if m.claude then
		table.insert(cards, card(m.claude, m.tabs))
	end
	for _, c in m.others do
		table.insert(cards, card(c, {}))
	end
	panel.render(ui.column({ gap = 10, padding = 10 }, {
		ui.row({ align = "center", gap = 8 }, {
			ui.label({ text = m.title, fontSize = 15, fontWeight = "bold", flexGrow = 1 }),
			ui.button({ glyph = "refresh", variant = "ghost", tooltip = m.refresh, enabled = not m.busy, onClick = onRefresh }),
		}),
		ui.scroll({ flexGrow = 1, gap = 10 }, cards),
	}))
end

function onRefresh()
	noctalia.state.set("cmd.refresh", (tonumber(noctalia.state.get("cmd.refresh")) or 0) + 1)
end

-- The panel opens on the active account, like v4.
function onOpen(_context: any)
	isOpen = true
	claudeTab = 1
	render()
end

function onClose()
	isOpen = false
end

noctalia.state.watch("view", function(v)
	current = v
	render()
end)
current = noctalia.state.get("view")
```

`card()` calls the global `render`, which is defined after it; that is fine in Luau because
the lookup happens at call time.
